




class Node {
  public:
    int data;
    Node* next;
    Node(int val) {
        data = val;
        next = nullptr;
    }
};

class myQueue {
  public:
    Node* head;
    Node* tail;
    int len;
    myQueue() {
        // Initialize your data members
        this->head=NULL;
        this->tail=NULL;
        this->len=0;   
    }
    bool isEmpty() {
        // check if the queue is empty
        if(len==0) return true;
        else return false;
    }
    void enqueue(int x) {
        // Adds an element x at the rear of the queue
        Node* temp=new Node(x);
        if(head==NULL) {
            head=temp;
            tail=head;
            len++;
        }
        else{
            tail->next=temp;
            tail=tail->next;
            len++;
        }
    }
    void dequeue() {
        // Removes the front element of the queue
        if(head!=NULL) {
            head=head->next;
            len--;
        }
    }
    int getFront() {
        // Returns the front element of the queue
        // If queue is empty, return -1
        if(head!=NULL) return head->data;
        else return -1;
    }

    int size() {
        // Returns the current size of the queue.
        return len;
    }
};
