



class Solution {
public:
    bool findRotation(vector<vector<int>>& mat, vector<vector<int>>& target) {
        for(int i=0;i<4;i++){
            int n=mat.size();
            for(int i=0;i<n;i++){
                for(int j=i;j<n;j++){
                    swap(mat[i][j],mat[j][i]);
                }
            }
            for(int k=0;k<n;k++){
                int i=0;
                int j=n-1;
                while(i<j){
                    swap(mat[k][i],mat[k][j]);
                    i++;
                    j--;
                }
            }
            if(mat==target) return true;
        }
        return false;
    }
};
