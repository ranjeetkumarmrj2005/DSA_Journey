




class Solution {
public:
    int factorial(int n, int r){
        int ans=1;
        for(int i=1;i<=r;i++){
            ans=ans*(n-i+1)/i;
        }
        return ans;
    }
    vector<vector<int>> generate(int numRows) {
        vector<vector<int>>ans;
        for(int i=0;i<numRows;i++){
            vector<int>v;
            for(int j=0;j<=i;j++){
                int x=factorial(i,j);
               v.push_back(x);
            }
            ans.push_back(v);
        }
        return ans;
    }
};
