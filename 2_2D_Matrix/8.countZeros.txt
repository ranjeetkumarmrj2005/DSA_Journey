


//Back-end complete function Template for C++

class Solution {
  public:
    int countZeros(vector<vector<int>>& mat) {
        int n=mat.size();
        int ans=0;
        int i=0;
        int j=n-1;
        while(i<n && j>=0){
            if(mat[i][j]==1){
                ans=ans+(n-i);
                j--;
            }
            else i++;
        }
        return n*n-ans;
    }
};
