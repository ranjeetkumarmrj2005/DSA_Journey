


class Solution {
  public:
    vector<int> matrixDiagonally(vector<vector<int>>& mat) {
        int n=mat.size();
        vector<int>v;
        int row = 0;
        int col = 0;
        bool upRight = true;
        for (int i = 0; i < n*n; i++) {
            v.push_back(mat[row][col]);
            if (upRight) {
                if (col == n - 1) { 
                    row++; 
                    upRight = false; 
                } 
                else if (row == 0) { 
                    col++; 
                    upRight = false; 
                } 
                else { 
                    row--; col++; 
                }
            } 
            else {
                if (row == n - 1) { 
                    col++; 
                    upRight = true; 
                } 
                else if (col == 0) { 
                    row++; 
                    upRight = true; 
                } 
                else { 
                    row++; col--;
                }
            }
        }
        return v;
    }
};
