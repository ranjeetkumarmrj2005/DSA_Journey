class Solution {
public:
    string repeatLimitedString(string s, int repeatLimit) {
        unordered_map<char,int>m;
        for(int i=0;i<s.length();i++){
            m[s[i]]++;
        }
        priority_queue< pair<char,int> >pq;
        for(auto &x:m){
            pq.push(x);
        }
        string result="";
        while(!pq.empty()){
            auto largest=pq.top();
            pq.pop();
            int len=min(largest.second,repeatLimit);
            for(int i=0;i<len;i++){
                result.push_back(largest.first);
            }
            largest.second=largest.second-len;
            // pair< char,int> secondLargest;
            if(largest.second>0){
                if(pq.empty()){
                    break;
                }
                auto secondLargest=pq.top();
                pq.pop();
                result+=secondLargest.first;
                secondLargest.second--;
                if(secondLargest.second>0){
                    pq.push(secondLargest);
                }
                pq.push(largest);        
            }
        }
        return result;
        
    }
};
