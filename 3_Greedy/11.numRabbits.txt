class Solution {
public:
    int numRabbits(vector<int>& answers) {
        int n=answers.size();
        unordered_map<int,int>m;
        for(int i=0;i<n;i++){
            m[answers[i]]++;
        }
        int ans=0;
        for(auto &x:m){
           int group=(x.second+x.first)/(x.first+1);
           ans=ans+group*(x.first+1);
        }
        return ans;
    }
};
