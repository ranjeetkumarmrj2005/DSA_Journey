


class Solution {
public:
    int minProd(vector<int>& arr) {
        int n = arr.size();
        if (n == 1) return arr[0];
        sort(arr.begin(), arr.end());
        if (arr[0] == 0) return 0;
        if (arr[0] > 0) return arr[0];

        int negcount = 0;
        for (int i = 0; i < n; i++) {
            if (arr[i] < 0) negcount++;
        }

        long long prod = 1;

        for (int i = 0; i < n; i++) {
            if (arr[i] == 0) continue;
            if (negcount % 2 == 0 && i == negcount - 1) {
                continue;
            }

            prod = (prod * arr[i]);
        }
        return prod;
    }
};
