

class Solution {
public:
    string makeFancyString(string s) {
        int n=s.size();
        string ans="";
        int i=0;
        int j=1;
        ans=ans+s[0];
        while(j<n){
            if(s[j]==s[i]){
                if((j-i)<2) ans.push_back(s[j]);
                j++;
            }
            else{
                i=j;
                ans.push_back(s[j]);
                j++;
            }
        }
        return ans;
    }
};
