


class Solution {
public:
    vector<int> sortArrayByParityII(vector<int>& nums) {
        int n=nums.size();
        int evenIdx=0;
        int oddIdx=1;
        vector<int>v(n);
        for(int i=0;i<n;i++){
            if(nums[i]%2==0){
                v[evenIdx]=nums[i];
                evenIdx+=2;
            }
            else{
                v[oddIdx]=nums[i];
                oddIdx+=2;
            }
        }
        return v;
    }
};
