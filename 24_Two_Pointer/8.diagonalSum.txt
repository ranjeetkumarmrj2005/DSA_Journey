


class Solution {
public:
    int diagonalSum(vector<vector<int>>& mat) {
        int n=mat.size();
        int i=0;
        int j=n-1;
        int sum=0;

        while(i<n){
            sum=sum+mat[i][i];
            if(i!=j) sum=sum+mat[i][j];
            i++;
            j--;
        }
        return sum;
    }
};
