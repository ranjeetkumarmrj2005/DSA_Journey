// ============================================================
// 1. VECTOR
// ============================================================

#include <vector>
using namespace std;

vector<int> v = {1, 2, 3};

v.push_back(4);          // Add at end — TC: O(1) amortized
v.pop_back();            // Remove last element — TC: O(1)

v.size();                // Number of elements — TC: O(1)
v.empty();               // Check if empty — TC: O(1)
v.clear();               // Remove all elements — TC: O(n)

v.front();               // First element — TC: O(1)
v.back();                // Last element — TC: O(1)

v.begin();               // Iterator to first element — TC: O(1)
v.end();                 // Iterator past last element — TC: O(1)

v.insert(v.begin() + 1, 10);  // Insert at index — TC: O(n)
v.erase(v.begin() + 1);       // Remove at index — TC: O(n)


// ============================================================
// 2. STRING
// ============================================================

#include <string>

string s = "hello";

s.size();                // Length — TC: O(1)
s.length();              // Length — TC: O(1)

s.empty();               // Check empty — TC: O(1)

s.push_back('a');        // Add character at end — TC: O(1)
s.pop_back();            // Remove last character — TC: O(1)

s.front();               // First character — TC: O(1)
s.back();                // Last character — TC: O(1)

s.substr(1, 3);          // Substring — TC: O(k), k = substring length

s.find("ll");            // Find substring — TC: O(n*m) worst case
s.find('l');             // Find character — TC: O(n)

s.erase(1, 2);            // Remove 2 chars — TC: O(n)

s.insert(1, "abc");       // Insert string — TC: O(n)

reverse(s.begin(), s.end()); // Reverse — TC: O(n)


// ============================================================
// 3. ALGORITHM
// ============================================================

#include <algorithm>

sort(v.begin(), v.end());            // Ascending — TC: O(n log n)
sort(v.rbegin(), v.rend());          // Descending — TC: O(n log n)

reverse(v.begin(), v.end());          // Reverse — TC: O(n)

swap(a, b);                           // Swap — TC: O(1)

min(a, b);                            // Minimum of 2 — TC: O(1)
max(a, b);                            // Maximum of 2 — TC: O(1)

min({a, b, c, d});                    // Minimum of multiple — TC: O(k)
max({a, b, c, d});                    // Maximum of multiple — TC: O(k)

*min_element(v.begin(), v.end());     // Minimum element — TC: O(n)
*max_element(v.begin(), v.end());     // Maximum element — TC: O(n)

find(v.begin(), v.end(), x);          // Find x — TC: O(n)

count(v.begin(), v.end(), x);         // Count x — TC: O(n)

binary_search(v.begin(), v.end(), x); // Check if exists — TC: O(log n)
// vector must be sorted

lower_bound(v.begin(), v.end(), x);   // First position >= x — TC: O(log n)
upper_bound(v.begin(), v.end(), x);   // First position > x — TC: O(log n)

next_permutation(v.begin(), v.end()); // Next permutation — TC: O(n)
prev_permutation(v.begin(), v.end()); // Previous permutation — TC: O(n)


// ============================================================
// 4. UNORDERED_MAP
// ============================================================

#include <unordered_map>

unordered_map<char, int> mp;

mp[key]++;                         // Increase frequency — TC: O(1) average
mp[key] = value;                   // Insert/update — TC: O(1) average

mp.find(key);                      // Find key — TC: O(1) average
mp.count(key);                     // Check existence — TC: O(1) average

mp.erase(key);                     // Remove key — TC: O(1) average

mp.size();                         // Number of keys — TC: O(1)
mp.empty();                        // Check empty — TC: O(1)

mp.clear();                        // Remove everything — TC: O(n)

mp.begin();                        // Iterator — TC: O(1)
mp.end();                          // Iterator — TC: O(1)


// ============================================================
// 5. MAP
// ============================================================

#include <map>

map<int, int> mp;

mp[key] = value;                   // Insert/update — TC: O(log n)

mp.find(key);                      // Find key — TC: O(log n)
mp.count(key);                     // Check existence — TC: O(log n)

mp.erase(key);                     // Remove key — TC: O(log n)

mp.size();                         // Number of keys — TC: O(1)
mp.empty();                        // Check empty — TC: O(1)

mp.clear();                        // Remove everything — TC: O(n)


// ============================================================
// 6. UNORDERED_SET
// ============================================================

#include <unordered_set>

unordered_set<int> st;

st.insert(x);                      // Insert — TC: O(1) average

st.find(x);                        // Find — TC: O(1) average

st.count(x);                       // Check existence — TC: O(1) average

st.erase(x);                       // Remove — TC: O(1) average

st.size();                         // Number of elements — TC: O(1)
st.empty();                        // Check empty — TC: O(1)

st.clear();                        // Remove everything — TC: O(n)


// ============================================================
// 7. SET
// ============================================================

#include <set>

set<int> st;

st.insert(x);                      // Insert — TC: O(log n)

st.find(x);                        // Find — TC: O(log n)

st.count(x);                       // Check existence — TC: O(log n)

st.erase(x);                       // Remove — TC: O(log n)

st.size();                         // Number of elements — TC: O(1)
st.empty();                        // Check empty — TC: O(1)

st.clear();                        // Remove everything — TC: O(n)

*st.begin();                       // Smallest element — TC: O(1)
*st.rbegin();                      // Largest element — TC: O(1)


// ============================================================
// 8. STACK
// ============================================================

#include <stack>

stack<int> st;

st.push(x);                        // Add — TC: O(1)
st.pop();                          // Remove top — TC: O(1)

st.top();                          // Top element — TC: O(1)

st.empty();                        // Check empty — TC: O(1)
st.size();                         // Number of elements — TC: O(1)


// ============================================================
// 9. QUEUE
// ============================================================

#include <queue>

queue<int> q;

q.push(x);                         // Add at back — TC: O(1)
q.pop();                           // Remove front — TC: O(1)

q.front();                         // First element — TC: O(1)
q.back();                          // Last element — TC: O(1)

q.empty();                         // Check empty — TC: O(1)
q.size();                          // Number of elements — TC: O(1)


// ============================================================
// 10. DEQUE
// ============================================================

#include <deque>

deque<int> dq;

dq.push_back(x);                   // Add at back — TC: O(1)
dq.push_front(x);                  // Add at front — TC: O(1)

dq.pop_back();                     // Remove back — TC: O(1)
dq.pop_front();                    // Remove front — TC: O(1)

dq.front();                        // First element — TC: O(1)
dq.back();                         // Last element — TC: O(1)

dq.size();                         // Number of elements — TC: O(1)
dq.empty();                        // Check empty — TC: O(1)


// ============================================================
// 11. PRIORITY QUEUE
// ============================================================

#include <queue>

// MAX HEAP
priority_queue<int> pq;

pq.push(x);                        // Insert — TC: O(log n)
pq.pop();                          // Remove maximum — TC: O(log n)
pq.top();                          // Maximum element — TC: O(1)

pq.size();                         // Number of elements — TC: O(1)
pq.empty();                        // Check empty — TC: O(1)


// MIN HEAP
priority_queue<int, vector<int>, greater<int>> pq;

pq.push(x);                        // Insert — TC: O(log n)
pq.pop();                          // Remove minimum — TC: O(log n)
pq.top();                          // Minimum element — TC: O(1)

// ============================================================
// 19. COMMON LOOP SHORTCUTS
// ============================================================

// Range-based loop
// Overall TC: O(n)

for(auto x : v) {
    cout << x;
}


// Reference
// Overall TC: O(n)

for(auto &x : v) {
    x++;
}


// Const reference
// Overall TC: O(n)

    cout << x;
}

// pop_back()       → O(1)
// substr()         → O(k)
// find()           → O(n*m) worst case
// erase()          → O(n)

// UNORDERED_MAP / UNORDERED_SET
// insert           → O(1) average
// find             → O(1) average
// erase            → O(1) average

// MAP / SET
// insert           → O(log n)
// find             → O(log n)
// erase            → O(log n)

// STACK
// push             → O(1)
// pop              → O(1)
// top              → O(1)

// QUEUE
// push             → O(1)
// pop              → O(1)
// front             → O(1)

// PRIORITY_QUEUE
// push             → O(log n)
// pop              → O(log n)
// top              → O(1)

// ALGORITHMS
// sort()           → O(n log n)
// reverse()        → O(n)
// find()           → O(n)
// count()          → O(n)
// binary_search()  → O(log n)
// lower_bound()    → O(log n)
// upper_bound()    → O(log n)
// min_element()    → O(n)
// max_element()    → O(n)
