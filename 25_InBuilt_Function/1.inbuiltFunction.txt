


// ============================================================
// 1. VECTOR
// ============================================================

#include <vector>
using namespace std;

vector<int> v = {1, 2, 3};

v.push_back(4);          // Add at end
v.pop_back();            // Remove last element

v.size();                // Number of elements
v.empty();               // true if empty
v.clear();               // Remove all elements

v.front();               // First element
v.back();                // Last element

v.begin();               // Iterator to first element
v.end();                 // Iterator past last element

v.insert(v.begin() + 1, 10);  // Insert 10 at index 1
v.erase(v.begin() + 1);       // Remove element at index 1


// ============================================================
// 2. STRING
// ============================================================

#include <string>

string s = "hello";

s.size();                // Length
s.length();              // Length

s.empty();               // Check empty

s.push_back('a');        // Add character at end
s.pop_back();            // Remove last character

s.front();               // First character
s.back();                // Last character

s.substr(1, 3);          // Substring starting at 1, length 3

s.find("ll");            // Find substring
s.find('l');             // Find character

s.erase(1, 2);           // Remove 2 chars starting from index 1

s.insert(1, "abc");      // Insert string at index 1

reverse(s.begin(), s.end());

// 3. ALGORITHM
// ============================================================

#include <algorithm>

sort(v.begin(), v.end());            // Ascending
sort(v.rbegin(), v.rend());          // Descending

reverse(v.begin(), v.end());         // Reverse

swap(a, b);                          // Swap

min(a, b);                           // Minimum of 2
max(a, b);                           // Maximum of 2

min({a, b, c, d});                   // Minimum of multiple
max({a, b, c, d});                   // Maximum of multiple

*min_element(v.begin(), v.end());    // Minimum element
*max_element(v.begin(), v.end());    // Maximum element

find(v.begin(), v.end(), x);         // Find x

count(v.begin(), v.end(), x);        // Count x

binary_search(v.begin(), v.end(), x); // Check if x exists
// vector must be sorted

lower_bound(v.begin(), v.end(), x);  // First position >= x
upper_bound(v.begin(), v.end(), x);  // First position > x

next_permutation(v.begin(), v.end());

prev_permutation(v.begin(), v.end());

unordered_map<char, int> mp;

mp[key]++;                         // Increase frequency

mp[key] = value;                   // Insert/update

mp.find(key);                      // Find key

mp.count(key);                     // 1 if exists, 0 otherwise

mp.erase(key);                     // Remove key

mp.size();                         // Number of keys

mp.empty();                        // Check empty

mp.clear();                        // Remove everything

mp.begin();
mp.end();
