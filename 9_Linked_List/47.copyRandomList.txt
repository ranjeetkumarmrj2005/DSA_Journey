


// // Definition for a Node.
// class Node {
// public:
//     int val;
//     Node* next;
//     Node* random;
    
//     Node(int _val) {
//         val = _val;
//         next = NULL;
//         random = NULL;
//     }
// };
class Solution {
public:
    Node* copyRandomList(Node* head) {
        Node* newHead= new Node(-1);
        Node* farzi=newHead;
        Node* temp=head;
        while(temp){
            Node* x= new Node(temp->val);
            farzi->next=x;
            farzi=farzi->next;
            temp=temp->next;
        }
        Node* a=head;
        Node* b=newHead->next;
        // step 2: Make a map of <original, duplicate>
        unordered_map<Node*, Node*>m;
        Node* tempa=a;
        Node* tempb=b;
        while(tempa!=NULL){
            m[tempa]=tempb;
            tempa=tempa->next;
            tempb=tempb->next;
        }
        for(auto x:m){
            Node* o = x.first;
            Node* d = x.second;
            if(o->random!=NULL){
                Node* oRandom=o->random;
                Node* dRandom=m[oRandom];
                d->random=dRandom;
            }
        }
        return b;
    }
};
