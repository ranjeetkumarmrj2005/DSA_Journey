



class Solution {
public:
    ListNode* removeNthFromEnd(ListNode* head, int n) {
        if(head->next==NULL && n==1) return NULL;
        ListNode* temp=head;
        int count=0;
        while(temp!=NULL){
            temp=temp->next;
            count++;
        }
        count=count-n;
        if(count==0) return head->next;
        temp=head;
        while(count>1){
            temp=temp->next;
            count--;
        }
        temp->next=temp->next->next;
        return head;
    }
};
