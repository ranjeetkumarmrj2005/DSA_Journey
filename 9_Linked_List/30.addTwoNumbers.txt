


class Solution {
public:
    ListNode* addTwoNumbers(ListNode* l1, ListNode* l2) {
        ListNode* newHead = new ListNode(-1);
        ListNode* temp = newHead;
        int carry = 0;

        while (l1 != NULL || l2 != NULL || carry != 0) {
            int x = carry;

            if (l1 != NULL) {
                x += l1->val;
                l1 = l1->next;
            }

            if (l2 != NULL) {
                x += l2->val;
                l2 = l2->next;
            }

            carry = x / 10;

            ListNode* farzi = new ListNode(x % 10);
            temp->next = farzi;
            temp = temp->next;
        }

        return newHead->next;
    }
};
