



class Solution {
public:
    ListNode* partition(ListNode* head, int x) {
        if(head==NULL) return head;
        ListNode* lessx=new ListNode(-1);
        ListNode* newHeadlessx=lessx;
        ListNode* otherx=new ListNode(-1);
        ListNode* newHeadotherx=otherx;
        ListNode* temp=head;
        while(temp!=NULL){
            ListNode* farzi=temp;
            temp=temp->next;
            farzi->next=NULL;
            if(farzi->val<x){
                lessx->next=farzi;
                lessx=lessx->next;
            }
            else{
                otherx->next=farzi;
                otherx=otherx->next;
            }
        }
        lessx->next=newHeadotherx->next;
        return newHeadlessx->next;
    }
};
