


class Solution {
  public:
    int lengthOfLoop(Node *head) {
        Node* slow=head;
        Node* fast=head;
        while(fast!=NULL && fast->next!=NULL){
            slow=slow->next;
            fast=fast->next->next;
            if(fast==slow) break;
        }
        if(fast==NULL || fast->next==NULL) return 0;
        int count=1;
        fast=fast->next;
        while(fast!=slow){
            fast=fast->next;
            count++;
        }
        return count;
    }
};
