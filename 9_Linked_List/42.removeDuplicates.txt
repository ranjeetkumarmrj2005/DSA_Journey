


/* Structure of a link list node
class Node {
  public:
    int data;
    Node* next;
    Node* prev;
    Node(int value) {
        data = value;
        next = nullptr;
        prev = nullptr;
    }
};
*/
class Solution {
  public:
    Node* removeDuplicates(Node* head) {
        Node* i=head;
        Node* j=head;
        while(j!=NULL){
            if(i->data==j->data){
                j=j->next;
            }
            else{
                i=i->next;
                i->data=j->data;
                j=j->next;
            }
        }
        Node* temp=i->next;
        i->next=NULL;
        if(temp!=NULL) temp->prev=NULL;
        return head;
    }
};
