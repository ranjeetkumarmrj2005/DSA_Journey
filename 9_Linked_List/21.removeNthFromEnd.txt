



class Solution {
public:
    ListNode* removeNthFromEnd(ListNode* head, int n) {
        if(head->next==NULL && n==1) return NULL;
        ListNode* slow= new ListNode(-1);
        slow->next=head;
        ListNode* fast=slow;
        while(n>0){
            fast=fast->next;
            n--;
        }
        if(fast->next==NULL) return head->next;
        while(fast->next!=NULL){
            fast=fast->next;
            slow=slow->next;
        }
        slow->next=slow->next->next;
        return head;
    }
};
