



class Solution {
public:
    ListNode* reverse(ListNode* head){
        ListNode* p=NULL;
        ListNode* c=head;
        ListNode* n=head;
        while(c){
            n=n->next;
            c->next=p;
            p=c;
            c=n;
        }
        return p;
    }
    ListNode* addTwoNumbers(ListNode* l1, ListNode* l2) {
        l1=reverse(l1);
        l2=reverse(l2);
        ListNode* newHead=new ListNode(-1);
        ListNode* temp=newHead;
        int divResult=0;
        while(l1!=NULL && l2!=NULL){
            int x=l1->val+l2->val+divResult;
            divResult=x/10;
            ListNode* farzi=new ListNode(x%10);
            temp->next=farzi;
            temp=temp->next;
            l1=l1->next;
            l2=l2->next;
        }
        while(l1!=NULL){
            int x=l1->val+divResult;
            divResult=x/10;
            ListNode* farzi=new ListNode(x%10);
            temp->next=farzi;
            temp=temp->next;
            l1=l1->next;
        }
        while(l2!=NULL){
            int x=l2->val+divResult;
            divResult=x/10;
            ListNode* farzi=new ListNode(x%10);
            temp->next=farzi;
            temp=temp->next;
            l2=l2->next;
        }
        if(divResult==1){
            ListNode* farzi=new ListNode(1);
            temp->next=farzi;
            temp=temp->next;
        }
        return newHead=reverse(newHead->next);
    }
};
