


/* Linked List Node Structure
class Node {
public:
    int data;
    Node* next;
    Node(int d) {
        data = d;
        next = nullptr;
    }
};
*/

class Solution {
  public:
    Node* arrayToList(vector<int>& nums) {
        int n=nums.size();
        Node* head=new Node(-1);
        Node* tail=head;
        for(int i=0;i<n;i++){
            Node* temp=new Node(nums[i]);
            tail->next=temp;
            tail=tail->next;
        }
        return head->next;
        
    }
};
