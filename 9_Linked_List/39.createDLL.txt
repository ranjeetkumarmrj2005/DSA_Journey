class Solution {
  public:
    Node* createDLL(vector<int>& nums) {
        Node* head=new Node(-1);
        Node* temp=head;
        int n=nums.size();
        int i=0;
        while(i<n){
            Node* farzi=new Node(nums[i]);
            temp->next=farzi;
            farzi->prev=temp;
            temp=temp->next;
            i++;
        }
        head=head->next;
        head->prev=NULL;
        return head;
    }
};
