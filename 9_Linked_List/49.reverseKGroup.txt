


class Solution {
public:
    ListNode* reverseKGroup(ListNode* head, int k) {
        if(head == NULL) return NULL;
        // Step 1: Check whether k nodes are available
        ListNode* temp = head;
        for(int i = 0; i < k; i++) {
            if(temp == NULL) return head;
            temp = temp->next;
        }
        // Step 2: Reverse the first k nodes
        ListNode* PREV = NULL;
        ListNode* CURR = head;
        ListNode* NEXT = head;
        int count = 0;
        while(CURR != NULL && count < k) {
            NEXT = NEXT->next;
            CURR->next = PREV;
            PREV = CURR;
            CURR = NEXT;
            count++;
        }
        // Step 3: Connect remaining reversed groups
        head->next = reverseKGroup(NEXT, k);
        // Step 4: Return new head
        return PREV;
    }
};
