
class Solution {
public:
    ListNode* deleteMiddle(ListNode* head) {
        if(head->next==NULL)return NULL;
        ListNode* slow=head;
        ListNode* fast=head;
        ListNode* temp1=new ListNode(-1);
        ListNode* temp2=new ListNode(-2);
        while(fast!=NULL && fast->next!=NULL){
            temp1=slow;
            slow=slow->next;
            temp2=slow->next;
            fast=fast->next->next;
        }
        temp1->next=temp2;
        return head;
    }
};
