



class Solution {
public:
    ListNode *detectCycle(ListNode *head) {
        unordered_map<ListNode*,int>m;
        ListNode* temp=head;
        int count=1;
        while(temp!=NULL){
            if(m.find(temp)!=m.end()) break;
            else {
                m[temp]=count++;
                temp=temp->next;
            }
        }
        return temp;
    }
};
