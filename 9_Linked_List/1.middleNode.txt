

class Solution {
public:
    ListNode* middleNode(ListNode* head) {
        int count=0;
        ListNode* temp=head;
        while(temp!=NULL){
            temp=temp->next;
            count++;
        }
        int x=count/2+1;
        temp=head;
        while(x>1){
            temp=temp->next;
            x--;
        }
        return temp;
    }
};
