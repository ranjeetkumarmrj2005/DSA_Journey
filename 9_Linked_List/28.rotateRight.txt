
class Solution {
public:
    ListNode* rotateRight(ListNode* head, int k) {
        if(head==NULL || head->next==NULL) return head;
        ListNode* temp=head;
        ListNode* tail;
        int count=1;
        while(temp->next!=NULL){
            temp=temp->next;
            count++;
        }
        tail=temp;
        k=k%count;
        if(k==0) return head;
        temp=head;
        count=count-k-1;
        while(count>0){
            temp=temp->next;
            count--;
        }
        
        ListNode* newHead=temp->next;
        temp->next=NULL;
        tail->next=head;
        return newHead;
    }
};
