


class Solution {
public:
    ListNode* rotateRight(ListNode* head, int k) {
        if(head==NULL || head->next==NULL||k==0) return head;
        ListNode* temp=head;
        int count=0;
        while(temp){
            temp=temp->next;
            count++;
        }
        k=k%count;
        if(k==0) return head;
        temp=head;
        count=count-k-1;
        while(count>0){
            temp=temp->next;
            count--;
        }
        ListNode* newHead=temp->next;
        ListNode* ans=newHead;
        while(newHead->next!=NULL){
            newHead=newHead->next;
        }
        temp->next=NULL;
        newHead->next=head;
        return ans;
    }
};
