



class Solution {
  public:
    int maximumSum(vector<vector<int>>& mat, int k) {
        int m=mat.size();
        int n=mat[0].size();
        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++){
                if(i==0 && j==0) continue;
                else if(i==0) mat[i][j]=mat[i][j]+mat[i][j-1];
                else if(j==0) mat[i][j]=mat[i][j]+mat[i-1][j];
                else mat[i][j]=mat[i][j]+mat[i][j-1]+mat[i-1][j]-mat[i-1][j-1];
            }
        }
        int maxSum=mat[k-1][k-1];
        for(int i=k-1;i<m;i++){
            for(int j=k-1;j<n;j++){
                int total=mat[i][j];
                if(i-k>=0) {
                    total=total-mat[i-k][j];
                }
                if(j-k>=0) {
                    total=total-mat[i][j-k];
                }
                if(i-k>=0 && j-k>=0){
                    total=total+mat[i-k][j-k];
                }
                maxSum=max(maxSum,total);
            }
        }
        return maxSum;
    }
};
