


class Solution {
public:
    vector<int> productExceptSelf(vector<int>& nums) {
        int n=nums.size();

        vector<int>left(n);
        left[0]=nums[0];
        for(int i=1;i<n;i++){
            left[i]=(left[i-1]*nums[i]);
        }

        vector<int>right(n);
        right[n-1]=nums[n-1];
        for(int i=n-2;i>=0;i--){
            right[i]=(nums[i]*right[i+1]);
        }

        vector<int>v(n);
        v[0]=right[1];
        v[n-1]=left[n-2];

        for(int i=1;i<n-1;i++){
            v[i]=(left[i-1]*right[i+1]);
        }
        return v;
    }
};
