

class NumMatrix {
public:
    vector<vector<int>>matrix;
    NumMatrix(vector<vector<int>>& matrix){
        int m=matrix.size();
        int n=matrix[0].size();
        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++){
                if(i==0 && j==0) continue;
                else if(i==0) matrix[i][j]=matrix[i][j]+matrix[i][j-1];
                else if(j==0) matrix[i][j]=matrix[i][j]+matrix[i-1][j];
                else matrix[i][j]=matrix[i][j]+matrix[i][j-1]+matrix[i-1][j]-matrix[i-1][j-1];
            }
        }
        this->matrix=matrix;
    }
    int sumRegion(int row1, int col1, int row2, int col2) {
        int total=matrix[row2][col2];
        if(col1>0){
            total=total-matrix[row2][col1-1];
        }
        if(row1>0){
            total=total-matrix[row1-1][col2];
        }
        if(row1>0 && col1>0){
            total=total+matrix[row1-1][col1-1];
        }
        return total;
    }
};
