


class Solution {
public:
    vector<int> corpFlightBookings(vector<vector<int>>& bookings, int n) {
        int m=bookings.size();
        vector<int>v(n);
        for(int i=0;i<m;i++){
            for(int j=bookings[i][0];j<=bookings[i][1];j++){
                v[j-1]=v[j-1]+bookings[i][2];
            }
        }
        return v;
    }
};
