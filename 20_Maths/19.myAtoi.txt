

class Solution {
public:
    int myAtoi(string s) {
        int n=s.size();
        int i=0;
        while(i<n && s[i]==' ') i++;
        if(i==n) return 0;

        int sign=1;
        if(s[i]=='+' || s[i]=='-'){
            sign=((s[i]=='-')?-1:1);
            i++;
        }
        int result=0;
        while(i<n && isdigit(s[i])){
            int digit= s[i]-'0';
            if(result>INT_MAX/10 || (result==INT_MAX/10 && digit>7)) {
                if(sign>0) return INT_MAX;
                else return INT_MIN;
            }
            result=result*10 + digit;
            i++;
        }
        return result*sign;
    }
};
