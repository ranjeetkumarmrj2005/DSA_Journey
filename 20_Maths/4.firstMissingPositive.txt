


bool cmp(int &a,int &b){
    if(a>0 && b>0) return a<b;
    if(a>0 && b<=0) return true;
    return false;
}

class Solution {
public:
    int firstMissingPositive(vector<int>& nums) {
        sort(nums.begin(),nums.end(),cmp);
        int first=1;
        int n=nums.size();
        for(int i=0;i<n;i++){
            if(i>0 && nums[i]==nums[i-1]) continue; 
            else if(nums[i]!=first) return first;
            else first++;
        }
        return nums[n-1]+1;
    }
};
