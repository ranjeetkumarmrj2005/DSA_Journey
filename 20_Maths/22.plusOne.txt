

class Solution {
public:
    vector<int> plusOne(vector<int>& nums) {
        int n=nums.size();
        vector<int>v;
        int i=n-1;
        while(i>=0 && nums[i]==9){
            v.push_back(0);
            i--;
        }
        if(i==-1) v.push_back(1);
        else {
            v.push_back(nums[i]+1);
            i--;
        }
        while(i>=0){
            v.push_back(nums[i]);
            i--;
        }
        reverse(v.begin(),v.end());
        return v;
    }
};
