

class Solution {
public:
    int mySqrt(int x) {
        if(x<=1) return x;
        int low=1;
        int high=x/2;
        while(low<=high){
            int mid=low+(high-low)/2;
            int sqroot=x/mid;
            if(mid==sqroot) return mid;
            else if(mid<sqroot) low=mid+1;
            else high=mid-1; 
        }
        return high;
    }
};
