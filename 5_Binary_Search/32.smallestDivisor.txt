class Solution {
public:
    int helper(vector<int>&nums,int mid){
        int n=nums.size();
        int count=0;
        for(int i=0;i<n;i++){
            double x=(double)nums[i]/mid;
            count=count+ceil(x);
        }
        return count;
    }
    int smallestDivisor(vector<int>& nums, int threshold) {
        int n=nums.size();
        int mx=INT_MIN;
        int low=1;
        int high=INT_MAX;
        int ans=-1;
        while(low<=high){
            int mid=low+(high-low)/2;
            int count=helper(nums,mid);
            if(count<=threshold) {
                ans=mid;
                high=mid-1;
            }
            else low=mid+1;
        }
        return ans;
    }
};
