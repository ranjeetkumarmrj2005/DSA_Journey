


class Solution {
  public:
    int nthRoot(int n, int m) {
        if(m>=0 && n==1) return m;
        if(m==1) return m;
        int low=1;
        int high=m/2;
        while(low<=high){
            int mid=low+(high-low)/2;
            long long x=1;
            for(int i=1;i<=n;i++){
                x=x*mid;
            }
            if(x==m) return mid;
            else if(x<m) low=mid+1;
            else high=mid-1;
        }
        long long t=1;
        for(int i=1;i<=n;i++){
            t=t*high;
        }
        if(t==m) return high;
        else return -1;
        
    }
};
