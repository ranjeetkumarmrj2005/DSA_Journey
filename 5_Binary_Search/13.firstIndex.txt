


class Solution {
  public:
    int firstIndex(vector<int> &arr) {
        int n=arr.size();
        int low=0;
        int high=n-1;
        int idx=-1;
        while(low<=high){
            int mid= low + (high-low)/2;
            if(arr[mid]!=1) low=mid+1;
            else {
                idx=mid;
                high=mid-1;
            }
        }
        return idx;
    }
};
