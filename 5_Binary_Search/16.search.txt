


class Solution {
public:
    int search(vector<int>& nums, int target) {
        int n=nums.size();
        int low=0;
        int high=n-1;
        while(low<high){
            int mid=low+(high-low)/2;
            if(nums[mid]>nums[high]) low=mid+1;
            else high=mid;
        }
        int pivoteIdx=low;
        if(target>=nums[pivoteIdx] && target<=nums[n-1]){
            low=pivoteIdx;
            high=n-1;
            while(low<=high){
                int mid=low+(high-low)/2;
                if(nums[mid]==target) return mid;
                else if(nums[mid]<target) low=mid+1;
                else high=mid-1;
            }
        }
        else{
            low=0;
            high=pivoteIdx-1;
            while(low<=high){
                int mid=low+(high-low)/2;
                if(nums[mid]==target) return mid;
                else if(nums[mid]<target) low=mid+1;
                else high=mid-1;
            }
        }
        return -1;
    }
};
