


class Solution {
  public:
    int helper(vector<int>&nums,long long mid){
        int n=nums.size();
        int count=1;
        long long currentpages=0;
        for(int i=0;i<n;i++){
            if(currentpages+nums[i]<=mid){
                currentpages=currentpages+nums[i];
            }
            else{
                count++;
                currentpages=nums[i];
            }
        }
        return count;
    }
    int findPages(vector<int> &nums, int k) {
        int n=nums.size();
        if(k>n) return -1;
        long long mx=INT_MIN;
        long long sum=0;
        for(int i=0;i<n;i++){
            mx=max(mx,(long long)nums[i]);
            sum=sum+nums[i];
        }
        long long ans=mx;
        long long low=mx;
        long long high=sum;
        while(low<=high){
            long long mid=low+(high-low)/2;
            if(helper(nums,mid)<=k){
                ans=mid;
                high=mid-1;
            }
            else low=mid+1;
        }
        return ans;
        
    }
};
