

class Solution {
  public:
    int lowerBound(vector<int>& arr, int target) {
        int n=arr.size();
        if(arr[n-1]<target) return n;
        int low=0;
        int high=n-1;
        int idx=-1;
        while(low<=high){
            int mid=low+(high-low)/2;
            if(arr[mid]>=target){
                idx=mid;
                high=mid-1;
            }
            else low=mid+1;
        }
        return idx;
        
    }
};
