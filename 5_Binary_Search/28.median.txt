

class Solution {
  public:
    int check(vector<vector<int>> &mat,int mid){
        int m=mat.size();
        int count=0;
        for(int i=0;i<m;i++){
            count=count+upper_bound(mat[i].begin(),mat[i].end(),mid)-mat[i].begin();
        }
        return count;
    }
    int median(vector<vector<int>> &mat) {
        int m=mat.size();
        int n=mat[0].size();
        int low=INT_MAX;
        int high=INT_MIN;
        for(int i=0;i<m;i++){
            low=min(low,mat[i][0]);
            high=max(high,mat[i][n-1]);
        }
        while(low<=high){
            int mid=low+(high-low)/2;
            int count=check(mat,mid);
            if(count<((m*n)/2)+1) low=mid+1;
            else high=mid-1;
        }
        return low;
    }
};
