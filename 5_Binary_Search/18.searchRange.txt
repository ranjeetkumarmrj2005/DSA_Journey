


class Solution {
public:
    vector<int> searchRange(vector<int>& nums, int target) {
        vector<int>v;
        int n=nums.size();
        int first=-1;
        int last=-1;
        int low=0;
        int high=n-1;
        while(low<=high){
            int mid=low+(high-low)/2;
            if(nums[mid]<=target){
                low=mid+1;
                if(nums[mid]==target) last=mid;
            }
            else high=mid-1;
        }
        low=0;
        high=n-1;
        while(low<=high){
            int mid=low+(high-low)/2;
            if(nums[mid]<target) low=mid+1;
            else {
                high=mid-1;
                if(nums[mid]==target) first=mid;
            }
        }
        return {first,last};
    }
};
