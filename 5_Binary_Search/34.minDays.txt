


class Solution {
public:
    int helper(vector<int>&nums,int mid,int k){
        int n=nums.size();
        int bqt=0;
        int count=0;
        for(int i=0;i<n;i++){
            if(nums[i]<=mid) {
                count++;
                if(count==k){
                    bqt++;
                    count=0;
                }
            }
            else count=0;
        }
        return bqt;
    }
    int minDays(vector<int>& nums, int m, int k) {
        int n=nums.size();
        if(n<(long long)m*k) return -1;
        int low=1;
        int high=INT_MAX;
        int ans=-1;
        while(low<=high){
            int mid=low+(high-low)/2;
            if(helper(nums,mid,k)>=m){
                ans=mid;
                high=mid-1;
            }
            else low=mid+1;
        }
        return ans;
    }
};
