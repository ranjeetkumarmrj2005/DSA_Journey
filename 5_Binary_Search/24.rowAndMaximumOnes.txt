

class Solution {
public:
    vector<int> rowAndMaximumOnes(vector<vector<int>>& mat) {
        int m=mat.size();
        int n=mat[0].size();
        int maxCount=0;
        int minIdx=0;
        for(int i=0;i<m;i++){
            sort(mat[i].begin(),mat[i].end());
            int low=0;
            int high=n-1;
            while(low<=high){
                int mid=low+(high-low)/2;
                if(mat[i][mid]==1) high=mid-1;
                else low=mid+1;
            }
            if(n-low>maxCount) {
                maxCount=n-low;
                minIdx=i;
            }
        }
        return {minIdx,maxCount};
    }
};
