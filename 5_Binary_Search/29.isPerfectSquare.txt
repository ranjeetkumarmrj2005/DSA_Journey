


class Solution {
public:
    bool isPerfectSquare(int num) {
        if(num<=1) return true;
        int low=1;
        int high=num/2;
        while(low<=high){
            int mid=low+(high-low)/2;
            if(mid==floor(((double)num/mid)) && mid==ceil((double)num/mid)) return true;
            else if(mid<num/mid) low=mid+1;
            else high=mid-1;
        }
        return false;
    }
};
