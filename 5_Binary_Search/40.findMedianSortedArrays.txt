class Solution {
public:
    double findMedianSortedArrays(vector<int>& nums1, vector<int>& nums2) {
        int n1 = nums1.size();
        int n2 = nums2.size();
        
        // Ensure nums1 is the smaller array to minimize the binary search range (O(log(min(n1, n2))))
        if (n1 > n2) return findMedianSortedArrays(nums2, nums1);
        
        int N = n1 + n2;     // Total length of both arrays combined
        int low = 0;         // Minimum elements we can pick from nums1
        int high = n1;       // Maximum elements we can pick from nums1
        
        while (low <= high) {
            int cut1 = low + (high - low) / 2; // Partition index for nums1
            int cut2 = N / 2 - cut1;           // Partition index for nums2 to balance left/right halves
            
            // Handle boundary conditions using INT_MIN and INT_MAX when cuts are at edges
            int l1 = (cut1 == 0) ? INT_MIN : nums1[cut1 - 1]; // Element to the left of cut1
            int l2 = (cut2 == 0) ? INT_MIN : nums2[cut2 - 1]; // Element to the left of cut2
            int r1 = (cut1 == n1) ? INT_MAX : nums1[cut1];     // Element to the right of cut1
            int r2 = (cut2 == n2) ? INT_MAX : nums2[cut2];     // Element to the right of cut2
            
            // Check if we have found the correct partition
            if (l1 <= r2 && l2 <= r1) {
                // If total elements (N) is odd, the median is the smallest element on the right side
                if (N % 2 != 0) return min(r1, r2);
                // If N is even, the median is the average of the max of the left side and min of the right side
                else return (double)(max(l1, l2) + min(r1, r2)) / 2.0;
            } 
            // If l1 is greater than r2, we have taken too many elements from nums1; move search space left
            else if (l1 > r2)  high = cut1 - 1; 
    
            // Otherwise, we have taken too few elements from nums1; move search space right
            else low = cut1 + 1;
        }
        return 0.0; // Fallback return for valid sorted inputs
    }
};
