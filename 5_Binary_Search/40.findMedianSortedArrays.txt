class Solution {
public:
    double findMedianSortedArrays(vector<int>& nums1, vector<int>& nums2) {
        // nums1=[1,3,8,9]
        // nums2=[2,4,5,6,7]
        int n1=nums1.size();
        int n2=nums2.size();
        if(n1>n2) return findMedianSortedArrays(nums2,nums1);
        int N=n1+n2;
        int low=0;
        int high=n1;
        while(low<=high){
            int cut1=low+(high-low)/2;
            int cut2=N/2-cut1;
            int l1=(cut1==0)?INT_MIN:nums1[cut1-1];
            int l2=(cut2==0)?INT_MIN:nums2[cut2-1];
            int r1=(cut1==n1)?INT_MAX:nums1[cut1];
            int r2=(cut2==n2)?INT_MAX:nums2[cut2];
            if(l1<=r2 && l2<=r1){
                if(N%2!=0) return min(r1,r2);  // if N is odd, only one median
                else return (double)(max(l1,l2)+min(r1,r2))/2.0; // if N is even , two median ,so take average
            }
            else if(l1>r2) high=cut1-1;
            else low=cut1+1;
        }
        return 0.0;
    }
};
