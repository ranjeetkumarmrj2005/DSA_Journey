


class Solution {
public:
    int helper(vector<int>&nums,int mid,int days){
        int n=nums.size();
        int temp=mid;
        int count=0;
        for(int i=0;i<n;i++){
            if(temp>=nums[i]) mid=mid-nums[i];
            else return INT_MAX;
            if(mid<0) {
                count++;
                mid=temp;
                mid=mid-nums[i];
                if(mid==0) count++;
            }
            
        }
        if(mid>0) return count+1;
        else return count;
    }
    int shipWithinDays(vector<int>& nums, int days) {
        int low=1;
        int high=INT_MAX;
        int ans=INT_MAX;
        while(low<=high){
            int mid=low+(high-low)/2;
            if(helper(nums,mid,days)<=days){
                ans=mid;
                high=mid-1;
            }
            else low=mid+1;
        }
        return ans;
    }
};
