class Solution {
public:
    int strStr(string haystack, string needle) {
        int n=haystack.size();
        int m=needle.size();
        int ans=-1;
        for(int i=0;i<n;i++){
            int a=0;
            if(haystack[i]==needle[a]){
                ans=i;
                while(i<n && a<m && haystack[i+a]==needle[a]){
                    a++;
                }
                if(a>=m) return ans;
            }
        }
        return -1;
    }
};
