

Given a valid parentheses string s, return the nesting depth of s. The nesting depth is the maximum number of nested parentheses.
Example 1:
Input: s = "(1+(2*3)+((8)/4))+1"
Output: 3


class Solution {
public:
    int maxDepth(string s) {
        int n=s.size();
        int count=0;
        int maxCount=0;
        for(int i=0;i<n;i++){
            if(s[i]=='('){
                count++;
                maxCount=max(maxCount,count);
            }
            if(s[i]==')'){
                count--;
            }
        }
        return maxCount;
    }
};
