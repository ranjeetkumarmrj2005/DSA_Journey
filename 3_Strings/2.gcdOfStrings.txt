

class Solution {
public:
    string gcdOfStrings(string str1, string str2) {
        int m=str1.size();
        int n=str2.size();
        if(str1+str2!=str2+str1) return "";
        
        int gcd=1;
        for(int i=1;i<=min(n,m);i++){
            if(n%i==0 && m%i==0){
                gcd=i;
            }
        }
        return str1.substr(0,gcd);
    }
};
