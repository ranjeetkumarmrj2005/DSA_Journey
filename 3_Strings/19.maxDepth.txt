

class Solution {
public:
    int maxDepth(string s) {
        int n=s.size();
        vector<int>dp(n,0);
        int count=0;
        for(int i=0;i<n;i++){
            if(s[i]=='('){
                count++;
            }
            if(s[i]==')'){
                count--;
            }
            dp[i]=count;
        }
        return *max_element(dp.begin(),dp.end());
    }
};
