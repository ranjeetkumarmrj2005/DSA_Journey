


class Solution {
public:
    int helper(char& ch){
        switch(ch){
            case 'I':return 1;
            case 'V':return 5;
            case 'X':return 10;
            case 'L':return 50;
            case 'C':return 100;
            case 'D':return 500;
            case 'M':return 1000;
            default: return 0;
        }
    }
    int romanToInt(string s) {
        int n=s.size();
        int i=n-1;
        int ans=0;
        int prev=0;
        while(i>=0){
            int temp=helper(s[i]);
            if(temp>=prev){
                ans=ans+temp;
                prev=temp;
                i--;
            }
            else{
                ans=ans-temp;
                prev=temp;
                i--;
            }
        }
        return ans;
    }
};
