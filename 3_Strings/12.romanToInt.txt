


class Solution {
public:
    int romanToInt(string s) {
        int n=s.size();
        int i=n-1;
        int ans=0;
        int prev=0;
        while(i>=0){
            int temp=0;
            if(s[i]=='I'){
                temp=1;
            }
            else if(s[i]=='V'){
                temp=5;
            }
            else if(s[i]=='X'){
                temp=10;
            }
            else if(s[i]=='L'){
                temp=50;
            }
            else if(s[i]=='C'){
                temp=100;
            }
            else if(s[i]=='D'){
                temp=500;
            }
            else{
                temp=1000;
            }
            if(temp>=prev){
                ans=ans+temp;
                prev=temp;
                i--;
            }
            else{
                ans=ans-temp;
                prev=temp;
                i--;
            }
        }
        return ans;
    }
};
