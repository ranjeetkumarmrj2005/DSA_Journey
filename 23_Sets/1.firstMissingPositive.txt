



class Solution {
public:
    int firstMissingPositive(vector<int>& nums) {
        unordered_set<int>s;  // for unordered_set every operation take O(1)  while for ordered_set it take O(nlog(n));
        for(int ele:nums){
            if(ele>0) s.insert(ele);
        }
        int first=1;
        while(s.size()>0){
            if(s.find(first)==s.end()) return first;
            else {
                s.erase(first);
                first++;
            }
        }
        return first;
    }
};
