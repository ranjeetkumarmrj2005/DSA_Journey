

class Solution {
  public:
    string postToPre(string s) {
        stack<string>st;
        int n=s.size();
        string ans="";
        for(int i=0;i<n;i++){
            if(isalnum(s[i])){
                st.push(string(1,s[i]));
            }
            else{
                string op2=st.top();
                st.pop();
                string op1=st.top();
                st.pop();
                string temp= s[i]+op1+op2;
                st.push(temp);
            }
        }
        return st.top();
        
        
    }
};
