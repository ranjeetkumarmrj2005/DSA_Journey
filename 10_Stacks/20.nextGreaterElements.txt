

class Solution {
public:
    vector<int> nextGreaterElements(vector<int>& nums) {
        int n=nums.size();
        stack<int>st;
        vector<int>v(n,-1);
        st.push(nums[n-1]);
        for(int i=2*n-1;i>=0;i--){
            int curr=nums[i%n];
            while(st.size()>0 && curr>=st.top()){
                st.pop();
            }
            if(i<n){
                if(st.size()>0) v[i]=st.top();
                else v[i]=-1;
            }
            st.push(curr);
        }
        return v;
    }
};
