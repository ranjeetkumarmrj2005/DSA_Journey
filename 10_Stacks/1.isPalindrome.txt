

class Solution {
public:
    bool isPalindrome(string s) {
        int n=s.size();
        stack<char>st;
        string ans="";
        for(int i=0;i<n;i++){
            if(isalnum(s[i])){
                char ch=tolower(s[i]);
                ans.push_back(ch);
                st.push(ch);
            }
        }
        for(auto &x:ans){
            if(x!=st.top()) return false;
            else st.pop();
        }
        return true;
    }
};
