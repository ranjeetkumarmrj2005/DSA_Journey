

class Solution {
public:

    int precedence(char ch) {
        if(ch == '^') return 3;
        if(ch == '*' || ch == '/') return 2;
        if(ch == '+' || ch == '-') return 1;
        return -1;
    }

    string infixToPostfix(string& s) {

        stack<char> st;
        string ans = "";

        for(int i = 0; i < s.size(); i++) {

            char ch = s[i];

            // Operand
            if(isalnum((unsigned char)ch)) {
                ans += ch;
            }

            // Opening bracket
            else if(ch == '(') {
                st.push(ch);
            }

            // Closing bracket
            else if(ch == ')') {

                while(!st.empty() && st.top() != '(') {
                    ans += st.top();
                    st.pop();
                }

                st.pop();
            }

            // Operators
            else {

                while(!st.empty() && st.top() != '(' &&
                    (precedence(st.top()) > precedence(ch) ||
                    (precedence(st.top()) == precedence(ch) && ch != '^'))) {

                    ans += st.top();
                    st.pop();
                }

                st.push(ch);
            }
        }

        // Pop remaining operators
        while(!st.empty()) {
            ans += st.top();
            st.pop();
        }

        return ans;
    }
};
