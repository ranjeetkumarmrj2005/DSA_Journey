

class Solution {
  public:
    vector<int> countGreater(vector<int> &nums, vector<int> &indices) {
        int n=nums.size();
        vector<int>v(n,0);
        stack< pair<int,int >st;
        st.push({nums[n-1],0});
        for(int i=n-2;i>=0;i--){
            int count=0;
            while(st.size()>0 && nums[i]<st.top()){
                count=count+st.top().second+1;
                st.pop();
            }
            v[i]=count;
            st.push({nums[i],count});
        }
    }
};
