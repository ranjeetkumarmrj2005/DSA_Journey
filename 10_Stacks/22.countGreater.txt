class Solution {
  public:
    vector<int> countGreater(vector<int> &nums, vector<int> &indices) {
        int n=nums.size();
        int m=indices.size();
        vector<int>v;
        for(int i=0;i<m;i++){
            int count=0;
            int idx=indices[i];
            int val=nums[idx];
            for(int j=idx+1;j<n;j++){
                if(nums[j]>val) count++;
            }
            v.push_back(count);
        }
        return v;
    }
};
