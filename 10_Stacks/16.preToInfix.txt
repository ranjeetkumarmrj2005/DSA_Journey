


class Solution {
  public:
    string preToInfix(string &s) {
        stack<string> st;

        for(int i = s.size()-1; i >= 0; i--) {
            char ch = s[i];

            if(isalnum(ch)) {
                st.push(string(1, ch));
            }
            else {
                string a = st.top();
                st.pop();

                string b = st.top();
                st.pop();

                string temp = "(" + a + ch + b + ")";
                st.push(temp);
            }
        }

        return st.top();
    }
};
