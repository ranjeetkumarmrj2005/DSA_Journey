


class Solution {
  public:
    vector<int> prevSmaller(vector<int>& nums) {
        int n=nums.size();
        stack<int>st;
        st.push(nums[0]);
        vector<int>v;
        v.push_back(-1);
        for(int i=1;i<n;i++){
            while(st.size()>0 && st.top()>=nums[i]){
                st.pop();
            } 
            if(st.size()==0) v.push_back(-1);
            else v.push_back(st.top());
            st.push(nums[i]);
        }
        return v;
    }
};
