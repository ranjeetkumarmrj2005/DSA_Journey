class Solution {
public:
    double minPrice(vector<int>& prices, vector<int>& discounts) {
        
        sort(prices.begin(),prices.end());
        sort(discounts.begin(),discounts.end());


        double sum=0;
        while(!prices.empty() && !discounts.empty()){
            int p=prices.back();
            prices.pop_back();
            int d=discounts.back();
            discounts.pop_back();
            sum=sum+ (p*(100.0-d))/100.0;
            
        }
        while(!prices.empty()){
            int p=prices.back();
            prices.pop_back();
            sum=sum+ p;
        }
        return sum;
    }
};
