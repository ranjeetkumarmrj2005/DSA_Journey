


class Solution {
public:
    int maxOperations(vector<int>& nums, int k) {
        int n=nums.size();
        unordered_map<int,int>m;
        for(int i=0;i<n;i++){
            m[nums[i]]++;
        }
        int count=0;
        for(int i=0;i<n;i++){
            int rem=k-nums[i];
            if(m.find(rem)!=m.end() && m.find(nums[i])!=m.end()){
                if(rem==nums[i]){
                    count=count+m[rem]/2;
                    m.erase(rem);
                }
                else{
                    m[nums[i]]--;
                    if(m[nums[i]]==0) m.erase(nums[i]);
                    m[rem]--;
                    if(m[rem]==0) m.erase(rem);
                    count++;
                }
            }
        }
        return count;
    }
};
