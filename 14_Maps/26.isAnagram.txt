


class Solution {
public:
    bool isAnagram(string s, string t) {
        if(s.length()!=t.length()) return false;
        unordered_map<char,int>s1;
        int n=s.length();
        for(int i=0;i<n;i++){
            s1[s[i]]++;
        }
        for(int i=0;i<n;i++){
            if(s1.find(t[i])!=s1.end()){
                s1[t[i]]--;
                if(s1[t[i]]==0){
                    s1.erase(t[i]);
                }
            }
            else return false;
        }
        if(s1.size()==0)return true;
        else return false;
    }
};
