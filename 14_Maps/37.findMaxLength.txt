

class Solution {
public:
    int findMaxLength(vector<int>& nums) {
        unordered_map<int, int> m;
        m[0]=-1;
        int n=nums.size();
        int sum=0;
        int count=0;
        for(int i=0;i<n;i++){
            if(nums[i]==0) sum--;
            else sum++;
            if(m.find(sum)!=m.end()) count=max(count,i-m[sum]);
            else m[sum]=i;
        }
        return count;
    }
};
