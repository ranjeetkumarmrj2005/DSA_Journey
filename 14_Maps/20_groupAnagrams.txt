class Solution {
public:
    vector<vector<string>> groupAnagrams(vector<string>& nums) {
        int n=nums.size();
        vector<vector<string>>v;
        unordered_map<string,vector<string>>m;
        for(int i=0;i<n;i++){
            string str=nums[i];
            sort(str.begin(),str.end());
            m[str].push_back(nums[i]);
        }
        for(auto &x:m){
            v.push_back(x.second);
        }
        return v;
    }
};
