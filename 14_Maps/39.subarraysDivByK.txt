


class Solution {
public:
    int subarraysDivByK(vector<int>& nums, int k) {
        int n=nums.size();
        unordered_map<int,int>m;
        m[0]=1;
        int sum=0;
        int count=0;
        for(int i=0;i<n;i++){
            sum=sum+nums[i];
            if(m.find((sum%k+k)%k)!=m.end()){
                count=count+m[(sum%k+k)%k];
            }
            m[(sum%k+k)%k]++;
        }
        return count;
    }
};
