


class Solution {
  public:
    int maxLength(vector<int>& nums) {
        int n=nums.size();
        unordered_map<int,int>m;
        int sum=0;
        int count=0;
        for(int i=0;i<n;i++){
            sum=sum+nums[i];
            if(sum==0) count=i+1; // or m[0]=-1;
            if(m.find(sum)!=m.end()) count=max(count,i-m[sum]);
            else m[sum]=i;
        }
        return count;
    }
};
