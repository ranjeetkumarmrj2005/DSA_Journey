

class Solution {
public:
    vector<int> topKFrequent(vector<int>& nums, int k) {
        int n=nums.size();
        unordered_map<int,int>m;
        for(int i=0;i<n;i++){
            m[nums[i]]++;
        }
        //priority_queue<int>pq; for the max heap, means max first
        //priority_queue< int, vector<int> ,greater<int> >pq; for the min heap, means min first
        // priority_queue< pair<int,int>,vector< pair<int,int>>,greater<pair<int,int>> > pq;
        typedef pair<int,int>T;
        priority_queue< T, vector<T>, greater<T> >pq;
        
        //priority_queue< pair<int,int>, vector<pair<int,int>>, greater<pair<int,int>> >pq;
        for(auto& x:m){
            pq.push({x.second,x.first});
            if(pq.size()>k) pq.pop();
        }
        vector<int>v;
        while(pq.size()>0){
            v.push_back(pq.top().second);
            pq.pop();
        }
        return v;
    }
};
