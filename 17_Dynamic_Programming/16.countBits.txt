




class Solution {
public:
    void helper(int n,vector<int>& dp){
        for(int i=0;i<=n;i++){
            int count=0;
            int j=i;
            while(j>0){
                int rem=j%2;
                j=j/2;
                if(rem==1) count++;
            }
            dp.push_back(count);
        }
    }
    vector<int> countBits(int n) {
        vector<int>dp;
        helper(n,dp);
        return dp;
    }
};
