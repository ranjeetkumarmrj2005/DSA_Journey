






class Solution {
  public:
    int helper(int n, vector<int>&dp){
        if(n<=3) return n-1;
        if(dp[n]!=-1) return dp[n];
        return dp[n]=(n-1)*((helper(n-1,dp)+helper(n-2,dp)));
        
    }
    int derangeCount(int n) {
        vector<int>dp(n+1,-1);
        return helper(n,dp);
    
        
    }
};
