class Solution {
public:
    int largestSumAfterKNegations(vector<int>& nums, int k) {
        int n=nums.size();
        priority_queue< int,vector<int>,greater<int> >pq;

        for(auto &ele:nums){
            pq.push(ele);
        }
        int pivot=k;
        for(int i=0;i<pivot;i++){
            if(pq.top()<0){
                int x=pq.top();
                pq.pop();
                pq.push(-x);
            }
            else if(pq.top()==0){
                break;
            }
            else{
                if(k%2!=0) {
                    int x=pq.top();
                    pq.pop();
                    pq.push(-x);
                }
                break;
            }
            k--;
        }
        int sum=0;
        while(pq.size()>0){
            sum=sum+pq.top();
            pq.pop();
        }
        return sum;
    }
};
