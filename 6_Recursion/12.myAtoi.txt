


class Solution {
  public:
    int myAtoi(string &s) {
        int n=s.size();
        int i=0;
        while(i<n && s[i]==' '){
            i++;
        }
        int sign=1;
        if(s[i]=='+' || s[i]=='-'){
            sign=(s[i]=='-'?-1:1);
            i++;
        }
        while(i<n && s[i]==0){
            i++;
        }
        long long finalAns=0;
        while(i<n && isdigit(s[i])){
            int digit=s[i]-'0';
            if( finalAns>INT_MAX/10 || (finalAns==INT_MAX/10 && s[i]>'7')){
                return (sign==1)?INT_MAX:INT_MIN;
            }
            else{
                finalAns=finalAns*10+digit;
                i++;
            }
        }
        return (sign==1)?finalAns:-1*finalAns;
    }
};
