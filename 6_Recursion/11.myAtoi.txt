


class Solution {
  public:
    int myAtoi(string &s) {
        int n=s.size();
        int i=0;
        while(i<n && s[i]==' '){
            i++;
        }
        int sign=1;
        if(s[i]=='+' || s[i]=='-'){
            sign=(s[i]=='-'?-1:1);
            i++;
        }
        while(i<n && s[i]==0){
            i++;
        }
        string ans="";
        while(i<n){
            if(s[i]>='0' && s[i]<='9'){
                ans.push_back(s[i]);
                i++;
            }
            else break;
        }
        int m=ans.size();
        long long finalAns=0;
        int j=0;
        while(j<m){
            int digit=ans[j]-'0';
            finalAns=finalAns*10+digit;
            j++;
        }
        if(finalAns>INT_MAX && sign==-1) return INT_MIN;
        else if(finalAns>INT_MAX && sign==1) return INT_MAX;
        else if(sign==-1) return -1*finalAns;
        else return finalAns;
    }
};
