class Solution {
public:
    int largestRectangleArea(vector<int>& nums) {
        int n=nums.size();
        stack< pair<int,int> >st;
        vector<int>rightSmaller(n,n);
        for(int i=n-1;i>=0;i--){
            while(st.size()>0 && nums[i]<=st.top().first){
                st.pop();
            }
            if(st.size()>0) rightSmaller[i]=st.top().second;
            else rightSmaller[i]=n;
            st.push({nums[i],i});
        }
        while(st.size()>0){
            st.pop();
        }
        vector<int>leftSmaller(n,-1);
        for(int i=0;i<n;i++){
            while(st.size()>0 && nums[i]<=st.top().first){
                st.pop();
            }
            if(st.size()>0) leftSmaller[i]=st.top().second;
            else leftSmaller[i]=-1;
            st.push({nums[i],i});
        }
        int area=0;
        for(int i=0;i<n;i++){
            area=max(area,nums[i]*(rightSmaller[i]-leftSmaller[i]-1));
        }
        return area;
    }
};
