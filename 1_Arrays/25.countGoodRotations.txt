

class Solution {
public:
    void helper(vector<int>& nums){
        int n=nums.size();
        int first=nums[0];
        for(int i=1;i<n;i++){
            nums[i-1]=nums[i];
        }
        nums[n-1]=first;
    }
    int countGoodRotations(vector<int>& nums) {
        int n=nums.size();
        int ans=0;
        long long sum1=0;
        long long sum2=0;
        for(int j=0;j<n;j++){
            if(j<n/2) sum1=sum1+(long long)nums[j];
            else sum2=sum2+(long long)nums[j];
        }
        if(sum1>sum2) ans++;
        int i=0;
        while(i<n-1){
            
            int icomingidx=(i+n/2)%n;
            int outgoing=nums[i];
            int incoming=nums[icomingidx];
            
            sum1=sum1-outgoing;
            sum1=sum1+(long long)incoming;

            sum2=sum2-incoming;
            sum2=sum2+(long long)outgoing;
            
            if(sum1>sum2) ans++;
            i++;
        }
        return ans;
    }
};
