


class Solution {
  public:
    int getSecondLargest(vector<int> &arr) {
        int first_max=-1;
        int second_max=-1;
        for(int i=0;i<arr.size();i++){
            if(arr[i]>first_max){
                second_max=first_max;
                first_max=arr[i];
            }
            else if(arr[i]!=first_max && arr[i]>second_max){
                second_max=arr[i];
            }
        }
        return second_max;
    }
};
