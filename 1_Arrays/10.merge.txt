


class Solution {
public:
    void merge(vector<int>& nums1, int m, vector<int>& nums2, int n) {
        int a,b,j;
        if(m >0 && n>0){
            j=m+n-1;
            a=m-1;
            b=n-1;
            while(a>=0 && b>=0){
                if(nums1[a]>=nums2[b]){
                    nums1[j]=nums1[a];
                    a--;
                }
                else{
                    nums1[j]=nums2[b];
                    b--;
                }
                j--;
            }
            if(b>=0){
                for(int i=0;i<=b;i++){
                    nums1[i]=nums2[i];
                }
            }
        }
        if( m==0 && n>0){
            for(int i=0;i<n;i++){
                nums1[i]=nums2[i];
            }
        }
    }
};
