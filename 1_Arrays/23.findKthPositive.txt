

class Solution {
public:
    int findKthPositive(vector<int>& arr, int k) {
        int n=arr.size();
        vector<int>v(2001,0);
        int count=0;
        for(int i=0;i<n;i++){
            v[arr[i]]=1;
        }
        for(int i=1;i<20001;i++){
            if(v[i]==0){
                count++;
                if(count==k) return i;
                
            }
        }
        return -1;
    }
};
