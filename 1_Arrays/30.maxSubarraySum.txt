


class Solution {
  public:
    int maxSubarraySum(vector<int>& nums, int k) {
        int n=nums.size();
        int sum=0;
        for(int i=0;i<k;i++){
            sum=sum+nums[i];
        }
        int maxsum=sum;
        int i=0;
        int j=k;
        while(j<n){
            sum=sum-nums[i];
            sum=sum+nums[j];
            maxsum=max(maxsum,sum);
            i++;
            j++;
        }
        return maxsum;
    }
};
