


class Solution {
public:
    int minSubArrayLen(int target, vector<int>& nums) {
        int n=nums.size();
        int i=0;
        int j=0;
        int minSize=n+1;
        int sum=0;
        while(i<n && j<n){
            sum=sum+nums[j];
            while(sum>=target){
                minSize=min(minSize,j-i+1);
                sum=sum-nums[i];
                i++;
            }
            j++;
        }
        if(minSize!=n+1) return minSize;
        else return 0;
    }
};
